const [actionLoading, setActionLoading] = useState(false);

const handleStatusChange = async (status) => {
    try {
      setActionLoading(true);

      await axios.put(
        `http://localhost:8000/api/student-application/update-status/${id}`,
        { status },
        {
          headers: {
            Authorization: `Bearer ${localStorage.getItem("token")}`,
          },
        }
      );

      setApp((prev) => ({ ...prev, status }));

      alert(`Application ${status.replaceAll("_", " ").toUpperCase()}`);
    } catch (error) {
      alert("Failed to update status");
    } finally {
      setActionLoading(false);
    }
  };





 <div className="action-buttons">
            <button
              className="btn-authority"
              disabled={actionLoading || app.status === "approved_by_authority"}
              onClick={() => handleStatusChange("approved_by_authority")}
            >
              Approve (Authority)
            </button>

            <button
              className="btn-faculty"
              disabled={actionLoading || app.status === "approved_by_faculty"}
              onClick={() => handleStatusChange("approved_by_faculty")}
            >
              Approve (Faculty)
            </button>

            <button
              className="btn-remove"
              disabled={actionLoading}
              onClick={() => handleStatusChange("removed")}
            >
              Remove
            </button>
          </div>
